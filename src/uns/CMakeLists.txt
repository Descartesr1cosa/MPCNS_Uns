# Each library only depends on the ones listed before it.
function(uns_add_lib name)
  add_library(${name} STATIC ${ARGN})
  uns_target_defaults(${name})
endfunction()

uns_add_lib(uns_core core/Params.cpp)
uns_add_lib(uns_complex  complex/Incidence.cpp complex/CellComplex.cpp complex/ComplexBuilder.cpp complex/Validate.cpp)
uns_add_lib(uns_geometry geometry/Geometry.cpp geometry/Reconstruct.cpp)
target_link_libraries(uns_geometry PUBLIC uns_complex)
uns_add_lib(uns_generate generate/HexMesh.cpp generate/BoxMesh.cpp generate/CubedSphereMesh.cpp)
target_link_libraries(uns_generate PUBLIC uns_complex)
uns_add_lib(uns_partition partition/CellGraph.cpp partition/Partitioner.cpp)
target_link_libraries(uns_partition PUBLIC uns_geometry)
uns_add_lib(uns_parallel parallel/Distributor.cpp)
target_link_libraries(uns_parallel PUBLIC uns_complex)
uns_add_lib(uns_io io/BinContainer.cpp io/MeshIO.cpp io/VtkWriter.cpp)
target_link_libraries(uns_io PUBLIC uns_parallel uns_complex)

# MPI-dependent library (only when MPI is available)
if(UNS_BUILD_MERCURY)
  find_package(MPI COMPONENTS CXX)
  if(MPI_CXX_FOUND)
    uns_add_lib(uns_comm comm/Exchanger.cpp)
    target_link_libraries(uns_comm PUBLIC uns_parallel MPI::MPI_CXX)
  endif()
endif()
uns_add_lib(uns_refine refine/RefineSpec.cpp refine/Lattice.cpp refine/RefineMesh.cpp refine/Indicators.cpp refine/Transfer.cpp)
target_include_directories(uns_refine PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(uns_refine PUBLIC uns_geometry uns_complex uns_core)
